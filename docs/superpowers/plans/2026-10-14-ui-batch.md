## Review Focus

1. **A remembered piece that is gone.** `lastPlayed` names a slug a later library build removed: the page must show "Start with Twinkle, Twinkle, Little Star" on First steps, never crash or show "Continue with undefined". Pinned by Task 1 (`startTarget` and `arrivalLevel` with a missing slug) and Task 2 (`a lastPlayed slug that is gone falls back to Start and First steps`).
2. **Odd search text.** Spaces only, punctuation only, capitals and accents (" ", "!!", "FÜR eli"): spaces only keep the level view, punctuation gives the empty result, accents and case are ignored. Pinned by Task 1 (`search.test.ts`, `libraryLayout` cases) and Task 3 (`Escape empties the field and brings back the chosen level`).
3. **A drag that carries no file.** A text selection or a link dragged inside the page must not show the drop frame or hide the page. Pinned by Task 3 (`a drag without files shows no frame`, unit and e2e).
4. **Pause at the edges.** Pausing in the count-in, then switching hands while paused, then rewinding, must never leave zero or two red roots. Pinned by Task 5 (`pause in the lead-in marks nothing`, `a hand switch while paused re-marks or clears`, and the e2e `paused in the count-in: nothing marked, Play red`).
5. **Degenerate pieces for the new rules.** A one-hand piece never shows letters; a piece whose hands are always within a fifth letters every bar; a one-bar piece's loop prefill is bar 1 to bar 1. Pinned by Task 6 (`letteredBars` one hand, always close) and Task 4 (`loopPrefill(1, 1)`).
6. **The letters on a phone.** At 360 px the keyboard scrolls and one hand can leave the view. The letters must still follow the score alone (spec 15, rule 5): no letter on a far-apart number, and no switching within a bar. Pinned by Task 6 (`hand-letters.spec.ts` plays Amazing Grace to its last bar on the mobile project, past the bars where a per-onset phone exception would have lettered it).

```python
def load_genres() -> list[dict]:
    """[{id, title}] in display order (catalog.json's genres table)."""
    return _read_catalog_file(CATALOG)["genres"]


def collection_genres() -> dict[str, str]:
    """collection id -> genre id; refuses a collection without a known genre."""
    known = {g["id"] for g in load_genres()}
    out: dict[str, str] = {}
    bad = []
    for c in load_collections():
        if c.get("genre") in known:
            out[c["id"]] = c["genre"]
        else:
            bad.append(f"{c['id']} ({c.get('genre')!r})")
    if bad:
        raise SystemExit(f"collection(s) without a known genre: {', '.join(bad)}")
    return out


def piece_genre(piece: dict, genres: dict[str, str]) -> str:
    """The piece's own genre when the catalog sets one, else its collection's."""
    return piece.get("genre") or genres[piece["collection"]]
```

```ts
/** Search over titles, composers and collections (spec section 6.3). */
export interface Searchable {
  title: string;
  composer: string;
  collection: string;
}

/** Lower case, accents removed: "Für Élise" -> "fur elise". */
export function foldText(s: string): string {
  return s.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase();
}

/** The folded words of a text, punctuation as separators. */
export function words(s: string): string[] {
  return foldText(s)
    .split(/[^\p{L}\p{N}]+/u)
    .filter((w) => w !== '');
}

/** Each piece's words, computed once per piece object. */
const haystacks = new WeakMap<object, string[]>();

function haystackOf(piece: Searchable): string[] {
  let h = haystacks.get(piece);
  if (h === undefined) {
    h = words(`${piece.title} ${piece.composer} ${piece.collection}`);
    haystacks.set(piece, h);
  }
  return h;
}

/** Every query word starts some word of the piece's title, composer or collection. */
export function matchesQuery(piece: Searchable, query: string): boolean {
  const terms = words(query);
  if (terms.length === 0) return false;
  const haystack = haystackOf(piece);
  return terms.every((t) => haystack.some((w) => w.startsWith(t)));
}

/** The matching pieces, in the order given (manifest order). */
export function searchPieces<T extends Searchable>(pieces: readonly T[], query: string): T[] {
  return pieces.filter((p) => matchesQuery(p, query));
}
```

```ts
/**
 * What the library page shows, as a pure function of the manifest and the reader's choices
 * (spec sections 5 and 6). The React views only render it.
 */
import {
  GENRE_MIN,
  GENRES_ABOVE,
  OTHER_BELOW,
  showsHeadings,
  showsPills,
  showsSearch,
} from './disclosure.js';
import { GENRES, type GenreId, type Level } from './levels.js';
import { searchPieces, words } from './search.js';

/** The fields of a manifest entry the library page reads. */
export interface LibraryEntry {
  slug: string;
  title: string;
  composer: string;
  level: Level;
  collection: string;
  collectionId: string;
  genre: GenreId;
  approxDurationSec: number;
}

export interface Row {
  slug: string;
  /** The title shown: shortened under its own collection heading (`shortTitle`). */
  title: string;
  /** The gray second line, or null when it would say nothing the page does not. */
  composerLine: string | null;
  /** The accessible name: `Open {full title} by {full composer}`. */
  label: string;
  duration: string;
}

/** A collection heading with its rows, "Other pieces", or a flat list (`heading` null). */
export interface Group {
  key: string;
  heading: string | null;
  rows: Row[];
}

/** One level of an all-levels list or of search results. */
export interface Section {
  level: Level;
  count: number;
  groups: Group[];
}

export type LibraryLayout =
  | { kind: 'all'; sections: Section[] }
  | { kind: 'level'; levels: Level[]; level: Level; count: number; groups: Group[] }
  | { kind: 'search'; query: string; count: number; sections: Section[] };

export const OTHER_PIECES = 'Other pieces';

/** 29 -> '0:29', 132.3 -> '2:12' (whole seconds, truncated). */
export function formatDuration(sec: number): string {
  const total = Math.max(0, Math.floor(sec));
  const minutes = Math.floor(total / 60);
  const seconds = total % 60;
  return `${minutes}:${String(seconds).padStart(2, '0')}`;
}

export function countText(n: number): string {
  return n === 1 ? '1 piece' : `${n} pieces`;
}

/** The composer up to the first " (": "Lowell Mason (tune Antioch)" -> "Lowell Mason". */
export function composerShort(composer: string): string {
  const i = composer.indexOf(' (');
  return (i < 0 ? composer : composer.slice(0, i)).trim();
}

/** "Traditional", "Traditional (…)" and "Anonymous": no name worth a line of its own. */
export function isUnnamed(composer: string): boolean {
  return /^(Traditional|Anonymous)\b/.test(composer);
}

/**
 * Under its own collection heading, a title ending in a parenthesis whose first comma-separated
 * part is made of words the heading already holds loses that part: "Melody (Album for the Young,
 * No. 1)" under "Schumann, Album for the Young, Op. 68" -> "Melody (No. 1)".
 */
export function shortTitle(title: string, heading: string): string {
  const m = /^(.*) \(([^()]*)\)$/.exec(title);
  if (!m) return title;
  const base = m[1]!;
  const parts = m[2]!.split(',').map((s) => s.trim());
  const first = words(parts[0]!);
  const said = new Set(words(heading));
  if (first.length === 0 || !first.every((w) => said.has(w))) return title;
  const rest = parts.slice(1).join(', ');
  return rest === '' ? base : `${base} (${rest})`;
}

/** "Für Elise (Bagatelle in A minor)" -> "Für Elise": for the Continue button. */
export function dropTrailingParenthesis(title: string): string {
  return title.replace(/\s*\([^()]*\)$/, '');
}

export function levelsIn(pieces: readonly LibraryEntry[]): Level[] {
  return [...new Set(pieces.map((p) => p.level))].sort((a, b) => a - b);
}

interface Draft {
  piece: LibraryEntry;
  title: string;
  /** Under a heading that names the group's one composer. */
  composerSaid: boolean;
}

interface DraftGroup {
  key: string;
  heading: string | null;
  drafts: Draft[];
}

function lastNameWord(composer: string): string | undefined {
  return words(composerShort(composer)).at(-1);
}

/** One list section's groups, in manifest order (spec 6.2). */
function draftGroups(pieces: readonly LibraryEntry[]): DraftGroup[] {
  const flat = (): DraftGroup[] => [
    {
      key: 'flat',
      heading: null,
      drafts: pieces.map((p) => ({ piece: p, title: p.title, composerSaid: false })),
    },
  ];
  if (!showsHeadings(pieces.length)) return flat();
  const byCollection = new Map<string, LibraryEntry[]>();
  for (const p of pieces) {
    const members = byCollection.get(p.collectionId);
    if (members) members.push(p);
    else byCollection.set(p.collectionId, [p]);
  }
  const groups: DraftGroup[] = [];
  const other: LibraryEntry[] = [];
  for (const [id, members] of byCollection) {
    if (members.length < OTHER_BELOW) {
      other.push(...members);
      continue;
    }
    const heading = members[0]!.collection;
    const composers = new Set(members.map((p) => composerShort(p.composer)));
    const only = composers.size === 1 ? members[0]!.composer : null;
    const last = only === null || isUnnamed(only) ? undefined : lastNameWord(only);
    const composerSaid = last !== undefined && words(heading).includes(last);
    groups.push({
      key: id,
      heading,
      drafts: members.map((p) => ({ piece: p, title: shortTitle(p.title, heading), composerSaid })),
    });
  }
  if (groups.length === 0) return flat();
  if (other.length > 0) {
    groups.push({
      key: 'other',
      heading: OTHER_PIECES,
      drafts: other.map((p) => ({ piece: p, title: p.title, composerSaid: false })),
    });
  }
  return groups;
}

/** Turns drafts into rows; a title shown twice anywhere in view keeps its composer on both rows. */
function finish(sections: DraftGroup[][]): Group[][] {
  const seen = new Map<string, number>();
  for (const groups of sections)
    for (const g of groups)
      for (const d of g.drafts) seen.set(d.title, (seen.get(d.title) ?? 0) + 1);
  return sections.map((groups) =>
    groups.map((g) => ({
      key: g.key,
      heading: g.heading,
      rows: g.drafts.map(({ piece, title, composerSaid }) => ({
        slug: piece.slug,
        title,
        composerLine:
          (seen.get(title) ?? 0) > 1
            ? composerShort(piece.composer)
            : isUnnamed(piece.composer) || composerSaid
              ? null
              : composerShort(piece.composer),
        label: `Open ${piece.title} by ${piece.composer}`,
        duration: formatDuration(piece.approxDurationSec),
      })),
    })),
  );
}

function sectionsOf(pieces: readonly LibraryEntry[]): Section[] {
  const levels = levelsIn(pieces);
  const members = levels.map((l) => pieces.filter((p) => p.level === l));
  const groups = finish(members.map(draftGroups));
  return levels.map((level, i) => ({ level, count: members[i]!.length, groups: groups[i]! }));
}

/** The page's list for the library, the chosen level and the search text (spec 6.2). */
export function libraryLayout(
  pieces: readonly LibraryEntry[],
  { level, query }: { level: Level | null; query: string },
): LibraryLayout {
  const q = query.trim();
  if (q !== '' && showsSearch(pieces.length)) {
    const found = searchPieces(pieces, q);
    return { kind: 'search', query: q, count: found.length, sections: sectionsOf(found) };
  }
  if (!showsPills(pieces.length)) return { kind: 'all', sections: sectionsOf(pieces) };
  const levels = levelsIn(pieces);
  const chosen = level !== null && levels.includes(level) ? level : levels[0]!;
  const inLevel = pieces.filter((p) => p.level === chosen);
  return {
    kind: 'level',
    levels,
    level: chosen,
    count: inLevel.length,
    groups: finish([draftGroups(inLevel)])[0]!,
  };
}

export interface StartTarget {
  piece: LibraryEntry;
  /** True when the piece is the reader's last one played (Continue), false for the first piece. */
  continuing: boolean;
}

/** The red button's piece: the last one played while it is in the library, else the first piece of the lowest level. */
export function startTarget(
  pieces: readonly LibraryEntry[],
  lastPlayed: string | null,
): StartTarget {
  const last = lastPlayed === null ? undefined : pieces.find((p) => p.slug === lastPlayed);
  if (last) return { piece: last, continuing: true };
  const lowest = levelsIn(pieces)[0];
  return { piece: pieces.find((p) => p.level === lowest)!, continuing: false };
}

export function startLabel(target: StartTarget): string {
  return target.continuing
    ? `Continue with ${dropTrailingParenthesis(target.piece.title)}`
    : `Start with ${target.piece.title}`;
}

/** The level the pills open on: chosen in this visit, else the last piece's, else the lowest. */
export function arrivalLevel(
  pieces: readonly LibraryEntry[],
  chosen: Level | null,
  lastPlayed: string | null,
): Level {
  const levels = levelsIn(pieces);
  if (chosen !== null && levels.includes(chosen)) return chosen;
  const last = lastPlayed === null ? undefined : pieces.find((p) => p.slug === lastPlayed);
  return last ? last.level : levels[0]!;
}

/** The genre pills of a level (spec 6.1): none unless it is long and two genres hold enough of it. */
export function genrePills(levelPieces: readonly LibraryEntry[]): GenreId[] {
  if (levelPieces.length <= GENRES_ABOVE) return [];
  const counts = new Map<GenreId, number>();
  for (const p of levelPieces) counts.set(p.genre, (counts.get(p.genre) ?? 0) + 1);
  const ids = GENRES.map((g) => g.id).filter((id) => (counts.get(id) ?? 0) >= GENRE_MIN);
  return ids.length >= 2 ? ids : [];
}
```

```tsx
/** True once `key` has differed from its first value: the list fades in on changes, never on arrival. */
function useChanged(key: string): boolean {
  const first = useRef(key);
  const changed = useRef(false);
  if (key !== first.current) changed.current = true;
  return changed.current;
}

/** What the polite live region says: the text, once, a moment after the last change, never on arrival. */
function useAnnouncement(text: string): string {
  const [said, setSaid] = useState('');
  const changed = useChanged(text);
  useEffect(() => {
    if (!changed) return;
    const timer = setTimeout(() => setSaid(text), ANNOUNCE_DELAY_MS);
    return () => clearTimeout(timer);
  }, [text, changed]);
  return said;
}
```

```ts
import { useEffect, useState } from 'react';

export const NOT_A_SCORE_REFUSAL =
  'That does not look like a MusicXML or MIDI file (.musicxml, .xml, .mxl, .mid or .midi).';

const SCORE_EXTENSIONS = ['.musicxml', '.xml', '.mxl', '.mid', '.midi'];

/** The file chooser's `accept` list. */
export const SCORE_ACCEPT = SCORE_EXTENSIONS.join(',');

export function isScoreFileName(name: string): boolean {
  const dot = name.lastIndexOf('.');
  return dot >= 0 && SCORE_EXTENSIONS.includes(name.slice(dot).toLowerCase());
}

/** Hands a MusicXML or MIDI file over as bytes; anything else, or a failed read, ends in the notice. */
export function acceptScoreFile(
  file: File | undefined,
  onFile: (name: string, bytes: Uint8Array) => void,
  onRefuse: (message: string) => void,
): void {
  if (!file) return;
  if (!isScoreFileName(file.name)) {
    onRefuse(NOT_A_SCORE_REFUSAL);
    return;
  }
  void file.arrayBuffer().then(
    (buffer) => onFile(file.name, new Uint8Array(buffer)),
    () => onRefuse(NOT_A_SCORE_REFUSAL),
  );
}

/** Whether a drag carries files (a text selection or a link dragged in the page does not). */
export function carriesFiles(dt: DataTransfer | null | undefined): boolean {
  return !!dt && Array.from(dt.types ?? []).includes('Files');
}

/**
 * The whole window takes a dropped file while the library is shown (spec 3, 7.6). Returns true
 * while files are dragged over the window: the page then shows the drop frame. Enter and leave
 * are counted, because moving onto a child element fires a leave for its parent.
 */
export function useWindowFileDrop(
  onFile: (name: string, bytes: Uint8Array) => void,
  onRefuse: (message: string) => void,
): boolean {
  const [dragging, setDragging] = useState(false);
  useEffect(() => {
    let depth = 0;
    const enter = (e: DragEvent) => {
      if (!carriesFiles(e.dataTransfer)) return;
      e.preventDefault();
      depth++;
      setDragging(true);
    };
    const over = (e: DragEvent) => {
      if (carriesFiles(e.dataTransfer)) e.preventDefault();
    };
    const leave = (e: DragEvent) => {
      if (!carriesFiles(e.dataTransfer)) return;
      depth = Math.max(0, depth - 1);
      if (depth === 0) setDragging(false);
    };
    const drop = (e: DragEvent) => {
      if (!carriesFiles(e.dataTransfer)) return;
      e.preventDefault();
      depth = 0;
      setDragging(false);
      acceptScoreFile(e.dataTransfer?.files?.[0], onFile, onRefuse);
    };
    window.addEventListener('dragenter', enter);
    window.addEventListener('dragover', over);
    window.addEventListener('dragleave', leave);
    window.addEventListener('drop', drop);
    return () => {
      window.removeEventListener('dragenter', enter);
      window.removeEventListener('dragover', over);
      window.removeEventListener('dragleave', leave);
      window.removeEventListener('drop', drop);
    };
  }, [onFile, onRefuse]);
  return dragging;
}
```

```ts
/**
 * When the keyboard letters the hands (T11, spec section 15): only in bars where the reader could
 * confuse them. Decided from the score once per piece, both hands, whatever is muted.
 */
import type { Hand, Note } from '@digitus2/engine';
import type { OnsetGroup } from './onsetGroups.js';

/** Hands this many semitones apart or closer are close: a perfect fifth, about four white keys. */
export const CLOSE_SEMITONES = 7;

/** The lowest right-hand key minus the highest left-hand key; null unless both hands sound. */
export function handGap(sounding: readonly { midi: number; hand: Hand }[]): number | null {
  let highL = -Infinity;
  let lowR = Infinity;
  for (const n of sounding) {
    if (n.hand === 'L') highL = Math.max(highL, n.midi);
    else lowR = Math.min(lowR, n.midi);
  }
  return Number.isFinite(highL) && Number.isFinite(lowR) ? lowR - highL : null;
}

/**
 * The measure indices to letter: every bar with an onset where the hands' sounding keys (struck
 * there, or held from earlier) are within CLOSE_SEMITONES or interleave, plus a lone bar between
 * two such bars, so the letters do not leave for one bar and come back.
 */
export function letteredBars(groups: readonly OnsetGroup[]): Set<number> {
  const close = new Set<number>();
  let held: Note[] = [];
  for (const g of groups) {
    held = held.filter((n) => n.offsetTick > g.tick);
    const gap = handGap([...g.notes, ...held]);
    if (gap !== null && gap <= CLOSE_SEMITONES) close.add(g.notes[0]!.measureIndex);
    held = held.concat(g.notes);
  }
  const out = new Set(close);
  for (const bar of close) if (!close.has(bar + 1) && close.has(bar + 2)) out.add(bar + 1);
  return out;
}

/** Whether the cursor's group (playing, or paused under T10) sits in a lettered bar. */
export function lettersShown(
  groups: readonly OnsetGroup[],
  currentGroup: number,
  bars: ReadonlySet<number>,
): boolean {
  const g = currentGroup >= 0 ? groups[currentGroup] : undefined;
  return g !== undefined && bars.has(g.notes[0]!.measureIndex);
}
```
